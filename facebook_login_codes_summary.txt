```python
# Cookie kaydetme fonksiyonu
def save_cookies(self, sb_instance, user_id):
    try:
        cookies = sb_instance.cdp.get_all_cookies()
        timestamp = int(time.time())
        filename = f"facebook_cookies_{timestamp}.json"
        
        with open(filename, 'w', encoding='utf-8') as f:
            json.dump(cookies, f, indent=2, ensure_ascii=False, default=str)
        
        return filename
    except Exception as e:
        return None

# Cookie yükleme fonksiyonu
def load_cookies(self, sb_instance, cookie_file):
    try:
        with open(cookie_file, 'r', encoding='utf-8') as f:
            cookies = json.load(f)
        
        for cookie in cookies:
            if isinstance(cookie, dict) and 'facebook.com' in cookie.get('domain', ''):
                sb_instance.cdp.set_all_cookies([cookie])
        
        return True
    except Exception as e:
        return False
```

```python
# Giriş başarısını kontrol etme
def check_login_success(self, sb_instance):
    try:
        current_url = sb_instance.cdp.get_current_url()
        page_title = sb_instance.cdp.get_title()
        
        # Giriş sayfası kontrolü
        login_indicators = ['giriş', 'login', 'kaydol', 'sign up']
        if any(indicator in page_title.lower() for indicator in login_indicators):
            return False
        
        # Ana sayfa elementlerini kontrol et
        home_elements = [
            '[data-testid="newsfeed"]',
            '[data-testid="left_nav_menu_list"]',
            'div[data-pagelet="LeftRail"]'
        ]
        
        for selector in home_elements:
            if sb_instance.cdp.is_element_visible(selector):
                return True
        
        return False
    except:
        return False
```

```python
def share_to_group(self, sb_instance, group_url, text, image_path=None):
    # Gruba git
    sb_instance.cdp.get(group_url)
    sb_instance.sleep(random.uniform(3, 5))
    
    # Paylaşım alanını bul
    post_selectors = [
        'span:contains("Bir şeyler yaz...")',
        'div[role="textbox"][aria-placeholder*="gönderi"]',
        'div[contenteditable="true"][role="textbox"]'
    ]
    
    for selector in post_selectors:
        if sb_instance.cdp.is_element_visible(selector):
            sb_instance.cdp.click(selector)
            break
    
    # Metni yaz
    if text:
        sb_instance.cdp.type(text)
    
    # Resim yükle (eğer varsa)
    if image_path:
        upload_selector = 'input[type="file"][accept*="image"]'
        sb_instance.cdp.choose_file(upload_selector, image_path)
    
    # Paylaş
    share_button = 'div:contains("Paylaş")[role="button"]'
    sb_instance.cdp.click(share_button)
```
